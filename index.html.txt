    function setBadge(target, text, type = "") {
      target.textContent = text;
      target.className = "status-badge" + (type ? " " + type : "");
    }

    function pushSample(tSec, ax, ay, az) {
      // Kombinierte Beschleunigungsmagnitude (ohne Gravitation via Hochpassfilter)
      const mag = Math.sqrt(ax * ax + ay * ay + az * az);

      // Langsamer Tiefpass (~0.6 Hz) entfernt grobe Hand-/Armdrift
      // Schneller Tiefpass (~18 Hz) glättet hochfrequentes Sensorrauschen
      const dt = state.lastSampleT ? Math.max(0.002, Math.min(0.05, tSec - state.lastSampleT)) : 0.016;
      state.lastSampleT = tSec;

      const alphaSlow = dt / (0.26 + dt);
      const alphaFast = dt / (0.012 + dt);

      state.lpSlow += alphaSlow * (mag - state.lpSlow);
      state.lpFast += alphaFast * (mag - state.lpFast);

      const bandPassed = state.lpFast - state.lpSlow;

      state.rawSamples.push({ t: tSec, mag });
      state.filteredSamples.push({ t: tSec, val: bandPassed });

      // Nur Daten der letzten 32 Sekunden behalten
      const cutoff = tSec - 32;
      while (state.filteredSamples.length > 0 && state.filteredSamples[0].t < cutoff) {
        state.filteredSamples.shift();
      }
      while (state.rawSamples.length > 0 && state.rawSamples[0].t < cutoff) {
        state.rawSamples.shift();
      }

      // Effektive Abtastrate schätzen
      if (state.filteredSamples.length > 10) {
        const span = state.filteredSamples[state.filteredSamples.length - 1].t - state.filteredSamples[0].t;
        if (span > 0.2) {
          const hz = Math.round((state.filteredSamples.length - 1) / span);
          el.sampleRate.value = hz;
        }
      }

      // Live-RMS der letzten Sekunde anzeigen
      const recent = state.filteredSamples.filter(s => s.t >= tSec - 1.0);
      if (recent.length > 0) {
        const meanSq = recent.reduce((acc, s) => acc + s.val * s.val, 0) / recent.length;
        el.rmsLive.textContent = "RMS: " + Math.sqrt(meanSq).toFixed(4);
      }
    }

    function handleMotionEvent(event) {
      if (state.demoMode) return;
      const acc = event.acceleration || event.accelerationIncludingGravity;
      if (!acc) return;
      const ax = acc.x || 0;
      const ay = acc.y || 0;
      const az = acc.z || 0;
      const tSec = performance.now() / 1000;
      pushSample(tSec, ax, ay, az);
    }

    async function requestSensorPermission() {
      if (typeof DeviceMotionEvent === "undefined") {

            return;

        el.startBtn.disabled = false;
        setBadge(el.sensorStatus, "Sensor aktiv", "ok");

      }
    }

    function resampleUniform(samples, targetHz = 50) {
      if (samples.length < 10) return [];
      const tStart = samples[0].t;
      const tEnd = samples[samples.length - 1].t;
      const duration = tEnd - tStart;
      if (duration < 5) return [];

      const n = Math.floor(duration * targetHz);
      const out = new Float64Array(n);
      let idx = 0;

      for (let i = 0; i < n; i++) {
        const t = tStart + i / targetHz;
        while (idx < samples.length - 2 && samples[idx + 1].t < t) {
          idx++;
        }
        const s0 = samples[idx];
        const s1 = samples[idx + 1] || s0;
        const span = Math.max(1e-6, s1.t - s0.t);
        const w = Math.max(0, Math.min(1, (t - s0.t) / span));
        out[i] = s0.val + w * (s1.val - s0.val);
      }

      // Mittelwert entfernen
      let mean = 0;
      for (let i = 0; i < n; i++) mean += out[i];
      mean /= n;
      for (let i = 0; i < n; i++) out[i] -= mean;

      return out;
    }

    function computeEnvelopeCorrelation(signal, fs, targetBpm) {
      // Vergleicht die Hüllkurve des Signals mit einer Referenz-Pulsfolge bei targetBpm
      const n = signal.length;
      if (n < fs * 5) return 0;

      const absSig = new Float64Array(n);
      // Glatte Hüllkurve (~0.25s Fenster)
      const win = Math.max(1, Math.round(fs * 0.25));
      for (let i = 0; i < n; i++) {
        let sum = 0;
        let cnt = 0;
        for (let j = Math.max(0, i - win); j <= Math.min(n - 1, i + win); j++) {
          sum += Math.abs(signal[j]);
          cnt++;
        }
        absSig[i] = sum / cnt;
      }

      const fHz = targetBpm / 60;
      const ref = new Float64Array(n);
      for (let i = 0; i < n; i++) {
        const t = i / fs;
        ref[i] = Math.pow(Math.max(0, Math.sin(Math.PI * fHz * t)), 6);
      }

      // Beste Phasenverschiebung über eine Periode suchen
      const periodSamples = Math.max(1, Math.round(fs / fHz));
      let bestCorr = -1;

      for (let shift = 0; shift < periodSamples; shift += Math.max(1, Math.floor(periodSamples / 24))) {
        let mA = 0, mB = 0;
        for (let i = 0; i < n; i++) {
          mA += absSig[i];
          mB += ref[(i + shift) % n];
        }
        mA /= n;
        mB /= n;

        let num = 0, denA = 0, denB = 0;
        for (let i = 0; i < n; i++) {
          const da = absSig[i] - mA;
          const db = ref[(i + shift) % n] - mB;
          num += da * db;
          denA += da * da;
          denB += db * db;
        }
        const den = Math.sqrt(denA * denB);
        if (den > 1e-12) {
          const r = num / den;
          if (r > bestCorr) bestCorr = r;
        }
      }

      return Math.max(0, bestCorr);
    }

    function analyzeMeasurement(watchBpm) {
      const fs = 50;
      const sig = resampleUniform(state.filteredSamples, fs);

      if (sig.length < fs * 8) {
        setBadge(el.signalBadge, "Zu wenig Daten", "bad");
        el.verdictBox.innerHTML = `
          <strong>Messung unvollständig</strong>
          Es wurden zu wenige Bewegungssamples erfasst. Bitte prüfe die Sensorfreigabe und halte das iPhone während der gesamten 30 Sekunden.
        `;
        return;
      }

      // RMS der gesamten Messung
      let sumSq = 0;
      for (let i = 0; i < sig.length; i++) sumSq += sig[i] * sig[i];
      const rms = Math.sqrt(sumSq / sig.length);

      // Goertzel / DFT-Spektrum im Bereich MIN_BPM .. MAX_BPM (in 0.5 BPM Schritten)
      const candidates = [];
      for (let bpm = MIN_BPM; bpm <= MAX_BPM; bpm += 0.5) {
        const freq = bpm / 60;
        const omega = (2 * Math.PI * freq) / fs;
        const coeff = 2 * Math.cos(omega);
        let q0 = 0, q1 = 0, q2 = 0;

        for (let i = 0; i < sig.length; i++) {
          q0 = coeff * q1 - q2 + sig[i];
          q2 = q1;
          q1 = q0;
        }
        const power = Math.sqrt(q1 * q1 + q2 * q2 - coeff * q1 * q2) / sig.length;
        candidates.push({ bpm, power });
      }

      // Durchschnittliche Spektralleistung als Rauschbasis
      const avgPower = candidates.reduce((a, b) => a + b.power, 0) / candidates.length;

      // Besten Peak finden
      let best = candidates[0];
      for (const c of candidates) {
        if (c.power > best.power) best = c;
      }

      // Leistung nahe der Apple Watch Herzfrequenz (±4 BPM) prüfen
      let nearWatch = candidates[0];
      for (const c of candidates) {
        if (Math.abs(c.bpm - watchBpm) <= 4 && c.power > nearWatch.power) {
          nearWatch = c;
        }
      }

      const snrPeak = avgPower > 1e-12 ? best.power / avgPower : 0;
      const snrWatch = avgPower > 1e-12 ? nearWatch.power / avgPower : 0;
      const envCorr = computeEnvelopeCorrelation(sig, fs, watchBpm);

      const diffBpm = Math.abs(best.bpm - watchBpm);
      const hasCorrelatedSignal =
        rms > 0.0008 &&
        ((diffBpm <= 4.5 && snrPeak >= 1.55) || (snrWatch >= 1.75 && envCorr >= 0.14));

      const confidenceScore = Math.min(
        99,
        Math.max(
          5,
          Math.round(
            (hasCorrelatedSignal ? 55 : 15) +
            Math.min(25, (snrWatch - 1) * 18) +
            Math.min(20, envCorr * 80) -
            Math.min(25, diffBpm * 1.5)
          )
        )
      );

      el.estHr.textContent = best.bpm.toFixed(1);
      el.refHr.textContent = String(watchBpm);
      el.corrMetric.textContent = `${confidenceScore}%`;
      el.corrSub.textContent = `SNR ${snrPeak.toFixed(2)} · Δ ${diffBpm.toFixed(1)} BPM`;

      if (hasCorrelatedSignal) {
        setBadge(el.signalBadge, "Signal korreliert", "ok");
        el.verdictBox.style.borderColor = "rgba(52, 211, 153, 0.45)";
        el.verdictBox.innerHTML = `
          <strong>Korrelierendes Signal erkannt</strong>
          Aus den Mikrobewegungen deiner Hand ergibt sich eine dominante Frequenz von <strong>${best.bpm.toFixed(1)} BPM</strong> (Apple Watch: <strong>${watchBpm} BPM</strong>, Abweichung: <strong>${diffBpm.toFixed(1)} BPM</strong>, Signal-Rausch-Verhältnis: <strong>${snrPeak.toFixed(2)}×</strong>). Unter ruhigen Bedingungen lässt sich hiermit ein zur Herzfrequenz passendes Bewegungs- bzw. Pulssignal feststellen.
        `;
      } else {
        setBadge(el.signalBadge, "Kein klares Signal", "warn");
        el.verdictBox.style.borderColor = "rgba(251, 191, 36, 0.45)";
        el.verdictBox.innerHTML = `
          <strong>Kein zuverlässig korrelierendes Signal gefunden</strong>
          Dominante Bewegungsfrequenz: <strong>${best.bpm.toFixed(1)} BPM</strong> gegenüber Apple Watch: <strong>${watchBpm} BPM</strong> (Abweichung: <strong>${diffBpm.toFixed(1)} BPM</strong>, RMS: <strong>${rms.toFixed(4)} m/s²</strong>). Überlagerungen durch Handhaltung, Griffstärke oder Umgebungsbewegung verdecken das schwache ballistokardiographische Signal.
        `;
      }
    }

    function renderLoop(nowMs) {
      const elapsedMs = state.recording ? Math.min(DURATION_MS, nowMs - state.startTime) : 0;
      const pct = state.recording ? (elapsedMs / DURATION_MS) * 100 : 0;

      el.progressFill.style.width = pct + "%";
      el.timerLabel.textContent = `${(elapsedMs / 1000).toFixed(1)} / 30.0 s`;

      if (state.recording && elapsedMs >= DURATION_MS) {
        state.recording = false;
        el.startBtn.disabled = false;
        el.permBtn.disabled = false;
        el.phaseLabel.textContent = "30s Messung abgeschlossen";
        const watchBpm = Math.max(35, Math.min(220, Number(el.watchHr.value) || 68));
        analyzeMeasurement(watchBpm);
      }

      // Wellenform zeichnen
      const w = el.canvas.width;
      const h = el.canvas.height;
      ctx.clearRect(0, 0, w, h);

      // Nulllinie
      ctx.strokeStyle = "rgba(148, 163, 184, 0.18)";
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(0, h / 2);
      ctx.lineTo(w, h / 2);
      ctx.stroke();

      const samples = state.filteredSamples;
      if (samples.length > 1) {
        const tEnd = samples[samples.length - 1].t;
        const windowSec = 6.0; // Letzte 6 Sekunden anzeigen
        const visible = samples.filter(s => s.t >= tEnd - windowSec);

        let maxAbs = 0.015;
        for (const s of visible) {
          const a = Math.abs(s.val);
          if (a > maxAbs) maxAbs = a;
        }
        // Sanfte Autokalibrierung
        maxAbs = Math.min(maxAbs, 0.5);

        ctx.strokeStyle = "#38bdf8";
        ctx.lineWidth = 2;
        ctx.beginPath();
        for (let i = 0; i < visible.length; i++) {
          const s = visible[i];
          const x = ((s.t - (tEnd - windowSec)) / windowSec) * w;
          const norm = Math.max(-1, Math.min(1, s.val / maxAbs));
          const y = h / 2 - norm * (h * 0.42);
          if (i === 0) ctx.moveTo(x, y);
          else ctx.lineTo(x, y);
        }
        ctx.stroke();
      }

      state.rafId = requestAnimationFrame(renderLoop);
    }

    el.permBtn.addEventListener("click", requestSensorPermission);
    el.startBtn.addEventListener("click", startMeasurement);
    el.demoBtn.addEventListener("click", toggleDemoMode);

    state.rafId = requestAnimationFrame(renderLoop);
